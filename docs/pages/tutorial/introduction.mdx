import React, { useState, useEffect } from 'react';
import {
  View,
  Text,
  FlatList,
  TextInput,
  TouchableOpacity,
  Linking,
  StyleSheet
} from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';

const shows = [
  {
    id: '1',
    title: 'Diriliş: Ertuğrul',
    platform: 'Netflix',
    link: 'https://www.netflix.com/title/80127001',
    subs: ['EN', 'AR', 'TR']
  },
  {
    id: '2',
    title: 'Masum',
    platform: 'BluTV',
    link: 'https://www.blutv.com/',
    subs: ['TR']
  },
  {
    id: '3',
    title: 'Fi',
    platform: 'YouTube',
    link: 'https://www.youtube.com/playlist?list=PLk4uE8Fv4LT4EtYlMk0GEfU7fTclGpC9U',
    subs: ['TR', 'EN']
  },
];

export default function App() {
  const [query, setQuery] = useState('');
  const [selectedSub, setSelectedSub] = useState(null);
  const [favorites, setFavorites] = useState([]);

  useEffect(() => {
    loadFavorites();
  }, []);

  const loadFavorites = async () => {
    const favs = await AsyncStorage.getItem('favorites');
    if (favs) {
      setFavorites(JSON.parse(favs));
    }
  };

  const toggleFavorite = async (id) => {
    const isFav = favorites.includes(id);
    const updated = isFav
      ? favorites.filter((favId) => favId !== id)
      : [...favorites, id];

    setFavorites(updated);
    await AsyncStorage.setItem('favorites', JSON.stringify(updated));
  };

  const filteredShows = shows.filter((show) => {
    const matchesQuery = show.title.toLowerCase().includes(query.toLowerCase());
    const matchesSub = selectedSub ? show.subs.includes(selectedSub) : true;
    return matchesQuery && matchesSub;
  });

  const renderItem = ({ item }) => {
    const isFav = favorites.includes(item.id);
    return (
      <View style={styles.card}>
        <Text style={styles.title}>{item.title}</Text>
        <Text style={styles.platform}>Watch on: {item.platform}</Text>
        <Text style={styles.subs}>Subtitles: {item.subs.join(', ')}</Text>

        <TouchableOpacity
          style={styles.button}
          onPress={() => Linking.openURL(item.link)}
        >
          <Text style={styles.buttonText}>Watch Now</Text>
        </TouchableOpacity>

        <TouchableOpacity onPress={() => toggleFavorite(item.id)}>
          <Text style={styles.favorite}>{isFav ? '❤️ Favorited' : '🤍 Add to Favorites'}</Text>
        </TouchableOpacity>
      </View>
    );
  };

  return (
    <View style={styles.container}>
      <Text style={styles.header}>🎬 Turkish Show Finder</Text>

      <TextInput
        style={styles.search}
        placeholder="Search shows..."
        placeholderTextColor="#aaa"
        value={query}
        onChangeText={setQuery}
      />

      <View style={styles.filterRow}>
        {['EN', 'AR', 'TR', null].map((lang) => (
          <TouchableOpacity
            key={lang ?? 'All'}
            style={[
              styles.filterButton,
              selectedSub === lang && styles.activeFilter
            ]}
            onPress={() => setSelectedSub(lang)}
          >
            <Text style={styles.filterText}>
              {lang ?? 'All'}
            </Text>
          </TouchableOpacity>
        ))}
      </View>

      <FlatList
        data={filteredShows}
        renderItem={renderItem}
        keyExtractor={(item) => item.id}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: '#111',
    paddingTop: 60,
    paddingHorizontal: 20,
  },
  header: {
    fontSize: 24,
    color: '#fff',
    marginBottom: 10,
    fontWeight: 'bold',
    textAlign: 'center'
  },
  search: {
    backgroundColor: '#222',
    color: '#fff',
    borderRadius: 8,
    padding: 10,
    marginBottom: 10
  },
  filterRow: {
    flexDirection: 'row',
    justifyContent: 'space-around',
    marginBottom: 15
  },
  filterButton: {
    backgroundColor: '#333',
    paddingVertical: 6,
    paddingHorizontal: 12,
    borderRadius: 6
  },
  activeFilter: {
    backgroundColor: '#ff3c3c'
  },
  filterText: {
    color: '#fff',
    fontWeight: 'bold'
  },
  card: {
    backgroundColor: '#222',
    padding: 20,
    borderRadius: 10,
    marginBottom: 15
  },
  title: {
    fontSize: 20,
    color: '#fff',
    fontWeight: 'bold'
  },
  platform: {
    color: '#ccc',
    marginTop: 5
  },
  subs: {
    color: '#aaa',
    marginTop: 5
  },
  button: {
    marginTop: 10,
    backgroundColor: '#ff3c3c',
    padding: 10,
    borderRadius: 8,
    alignItems: 'center'
  },
  buttonText: {
    color: '#fff',
    fontWeight: 'bold'
  },
  favorite: {
    color: '#f9c2ff',
    textAlign: 'center',
    marginTop: 10
  }
});
